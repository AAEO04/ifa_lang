    Audio Player (Console)

    Playlist Management with Ifá-Lang's Irosu and Odi Domains

    Overview

    Ifá-Lang does not yet have a native audio playback domain. This example implements a console-based audio player interface — managing a playlist with Odi file I/O, displaying metadata with Irosu, and using Ogunda for list operations. Audio playback hardware control is planned for a future release.

    Domains Used

    • Irosu - Console display, input reading, logging
    • Odi - ka (read playlist file), ko (write playlist), wa (check exists)
    • Ogunda - iwọn (len), maapu (map), yan (filter), fi (push/pop), yi_pada (map transform)
    • Ika - gigun (len), ge (slice), oruko_html (title helper)
    • Oyeku - sun (sleep simulating playback), jade (exit)

    Console Player Loop

    Simulated Playback 
    ese play_track(track_name) {
    Irosu.fo("Now playing: " + track_name);
    Irosu.fo("Duration: 3:45");
    Irosu.fo("--- [playing] ---");
    Oyeku.sun(2000); // simulate 2s of playback
    Irosu.fo("Track ended");
}

ese player_loop() {
    Irosu.fo("Audio Player v0.1 (console mode)");
    Irosu.fo("Commands: play <name>, list, add <name>, stop, exit");

    nigba (otito) {
        ayanmo cmd = Irosu.gbo("player> ");

        ti (cmd == "exit" or cmd == "quit") {
            Irosu.fo("Goodbye!");
            Oyeku.jade(0);
        }

        ti (Ika.gigun(cmd) == 0) { tesiwaju; }
        ti (cmd == "list") {
            Irosu.fo("Queue: [track1.mp3, track2.mp3, track3.mp3]");
        } bibẹkọ ti (cmd == "stop") {
            Irosu.fo("Stopped");
        } bibẹkọ {
            ti (Ika.ge(cmd, 0, 4) == "play") {
                ayanmo name = Ika.ge(cmd, 5, Ika.gigun(cmd));
                play_track(name);
            } bibẹkọ {
                Irosu.kigbe("Unknown command");
            }
        }
    }
}

player_loop();

    Track Queue with Ogunda

    Adding and Removing Tracks 
    ese add_track(queue, track) {
    ayanmo updated = queue;
    Ogunda.fi(updated, track);
    Irosu.fo("Added: " + track);
    pada updated;
}

ese skip_track(queue) {
    ayanmo remaining = queue;
    Ogunda.mu(remaining); // pop first
    Irosu.fo("Skipped to next track");
    pada remaining;
}

ayanmo queue = ["song1.ogg", "song2.ogg", "song3.ogg"];
queue = add_track(queue, "song4.ogg");
Irosu.fo("Queue size: " + Ogunda.iwọn(queue));
queue = skip_track(queue);

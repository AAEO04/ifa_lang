System Interface

CLI args, environment, and process control with Ogbe & Oyeku

Initialization

Init
Ogbe.bi();                   // initialize system environment

Command-Line Arguments (Ogbe)

Access arguments passed to your program:

CLI Args 
// Run with: ifa run app.ifa -- arg1 arg2 arg3

// Get all arguments
ayanmo args = Ogbe.awon_ohun();
Irosu.fo(args);  // ["arg1", "arg2", "arg3"]

// Get argument count
ayanmo count = Ogbe.iye_ohun(args);
Irosu.fo("Count: " + count);  // 3

// Get specific argument
ayanmo first = Ogbe.ohun(args, 0);
Irosu.fo(first);  // "arg1"
// Run with: ifa run app.ifa -- arg1 arg2 arg3

// Get all arguments
let args = Ogbe.args();
Irosu.println(args);  // ["arg1", "arg2", "arg3"]

// Get argument count
let count = Ogbe.iye_ohun(args);
Irosu.println("Count: " + count);  // 3

// Get specific argument
let first = Ogbe.ohun(args, 0);
Irosu.println(first);  // "arg1"

Environment Variables

Environment 
// Get environment variable
ayanmo path = Ogbe.ayika("PATH");
Irosu.fo("PATH: " + path);

// Get with default fallback
ayanmo home = Ogbe.ayika_tabi("HOME", "/default");
Irosu.fo("HOME: " + home);
// Get environment variable
let path = Ogbe.env("PATH");
Irosu.println("PATH: " + path);

// Get with default fallback
let home = Ogbe.env("HOME", "/default");
Irosu.println("HOME: " + home);

System Information

System Info 
// OS name
ayanmo os = Ogbe.eto();
Irosu.fo("OS: " + os);  // "linux", "windows", "macos"

// Home directory
ayanmo home = Ogbe.ile();
Irosu.fo("Home: " + home);

// Current working directory
ayanmo cwd = Ogbe.oju_ona();
Irosu.fo("CWD: " + cwd);

// Or use English alias
ayanmo cwd2 = Ogbe.cwd();
Irosu.fo("CWD: " + cwd2);

Process Exit (Oyeku)

Exit 
// Exit with specific code
Oyeku.jade(0);       // success
Oyeku.jade(1);       // error

// Shortcuts
Oyeku.jade(0);   // exit(0) - exit well

Oyeku.jade(1);   // exit(1) - exit badly
// Garbage collect / cleanup resources
Oyeku.gbale();

// Abort immediately (no cleanup)
Oyeku.da_duro();     // abort
// Exit with specific code
Oyeku.exit(0);        // success
Oyeku.exit(1);        // error

// Shortcuts
Oyeku.exit_ok();      // exit(0) - exit well
Oyeku.exit_err();     // exit(1) - exit badly

// Garbage collect / cleanup resources
Oyeku.gc();

// Abort immediately (no cleanup)
Oyeku.abort();        // abort

Sleep

Sleep 
// Sleep for seconds
Irosu.fo("Waiting...");
Oyeku.sun(2);                // 2 seconds
Irosu.fo("Done waiting");

// Sleep for milliseconds
Oyeku.sun_ms(500);           // 500 milliseconds

// Wait/pause
Oyeku.duro();                // indefinite pause

Practical: CLI Tool Pattern

CLI App 
isodale main() {
    ayanmo args = Ogbe.awon_ohun();
    ayanmo cmd = Ogbe.ohun(args, 0);
    
    yan (cmd) {
        "hello" => Irosu.fo("Hello, " + Ogbe.ayika_tabi("USER", "World"));
        "time"  => Irosu.fo(Iwori.bayi());
        "fail"  => Oyeku.jade(1);
        _       => Irosu.fo("Usage: app [hello|time|fail]");
    }
}

main();
fn main() {
    let args = Ogbe.args();
    let cmd = Ogbe.ohun(args, 0);
    
    match (cmd) {
        "hello" => Irosu.println("Hello, " + Ogbe.env("USER", "World"));
        "time"  => Irosu.println(Iwori.now());
        "fail"  => Oyeku.exit(1);
        _       => Irosu.println("Usage: app [hello|time|fail]");
    }
}

main();

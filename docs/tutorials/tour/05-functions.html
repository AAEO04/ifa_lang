Functions

Defining and calling functions

Defining Functions (isọdálẹ)

Use isọdálẹ (Yoruba: "definition") or fn to define functions:

Function Definition 
isọdálẹ greet(name) {
    Irosu.fo("Hello, " + name + "!");
}

// Calling a function
greet("Ifá");
greet("Ọba");
fn greet(name) {
    Irosu.println("Hello, " + name + "!");
}

// Calling a function
greet("Ifá");
greet("Ọba");

Return Values

Functions return the last expression by default:

Return Values 
isọdálẹ add(a, b) {
    a + b
}

isọdálẹ square(x) {
    x * x
}

ayanmo sum = add(5, 3);     // 8
ayanmo sq = square(4);      // 16
Irosu.fo("Sum: " + sum);     // Sum: 8
Irosu.fo("Squared: " + sq); // Squared: 16
fn add(a, b) {
    a + b
}

fn square(x) {
    x * x
}

let sum = add(5, 3);     // 8
let sq = square(4);      // 16
Irosu.println("Sum: " + sum);     // Sum: 8
Irosu.println("Squared: " + sq); // Squared: 16

Anonymous Functions (Closures)

Use the Yoruba arrow { args -> body } syntax:

Anonymous Functions 
// Anonymous function assigned to a variable
ayanmo double = { x -> x * 2 };

Irosu.fo(double(5));  // 10

// Passing a closure to a function
isọdẹlẹ apply(fn, value) {
    fn(value)
}

ayanmo result = apply({ x -> x + 1 }, 10);
Irosu.fo(result);  // 11
// Anonymous function assigned to a variable
let double = { x -> x * 2 };

Irosu.println(double(5));  // 10

// Passing a closure to a function
fn apply(fn, value) {
    fn(value)
}

let result = apply({ x -> x + 1 }, 10);
Irosu.println(result);  // 11
Syntax Note

Ifá-Lang uses { args -> body } for closures. The -> arrow separates parameters from the function body.

Closures Capturing Variables

Capturing Variables 
ayanmo multiplier = 3;
ayanmo triple = { x -> x * multiplier };

Irosu.fo(triple(5));  // 15
let multiplier = 3;
let triple = { x -> x * multiplier };

Irosu.println(triple(5));  // 15

Default Parameter Values

Default Parameters 
isọdẹlẹ greet(name, greeting = "Hello") {
    Irosu.fo(greeting + ", " + name + "!")
}

greet("Ifá");               // Hello, Ifá!
greet("Ọba", "Ẹ kú àbọ̀");  // Ẹ kú àbọ̀, Ọba!
fn greet(name, greeting = "Hello") {
    Irosu.println(greeting + ", " + name + "!")
}

greet("Ifá");               // Hello, Ifá!
greet("Ọba", "Welcome");    // Welcome, Ọba!

Named Arguments

Named Arguments 
isọdẹlẹ create_user(name, age, active = otito) {
    Irosu.fo(name + " is " + age + " active:" + active)
}

create_user(active: iro, name: "Ifá", age: 1000);
fn create_user(name, age, active = true) {
    Irosu.println(name + " is " + age + " active:" + active)
}

create_user(active: false, name: "Ifá", age: 1000);
